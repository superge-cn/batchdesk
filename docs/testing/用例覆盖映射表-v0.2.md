## 4. 复算与再生成

```bash
# 行数与唯一实例 ID 必须等于 TC v0.3 §3.1 的 296
python3 - <<'EOF'
import csv, collections
rows = list(csv.DictReader(open('docs/testing/用例覆盖映射表-v0.2.csv', encoding='utf-8')))
assert len(rows) == 296, len(rows)
assert len({r['实例ID'] for r in rows}) == 296
print(collections.Counter(r['状态'] for r in rows))
EOF
```

```bash
# 锚点实名校验：CSV 里写的每个测试名都必须在源码里存在，且不允许省略式写法
python3 - <<'EOF'
import csv, re
rows = list(csv.DictReader(open('docs/testing/用例覆盖映射表-v0.2.csv', encoding='utf-8')))
def fns(path):
    return set(re.findall(r'(?m)^\s*fn ([a-z0-9_]+)\(\)', open(path, encoding='utf-8').read()))
ga, af, cor = fns('crates/kernel/src/generic_apply.rs'), fns('crates/cli/tests/apply_flow.rs'), fns('crates/kernel/tests/corpus.rs')
bad = []
for r in rows:
    cell = r['自动化锚点']
    bad += [(r['实例ID'], n) for n in re.findall(r'generic_apply::tests::([a-z0-9_]+)', cell) if n not in ga]
    bad += [(r['实例ID'], n) for n in re.findall(r'apply_flow::([a-z0-9_]+)', cell) if n not in af]
    bad += [(r['实例ID'], n) for n in re.findall(r'corpus\.rs::([a-z0-9_]+)', cell) if n not in cor]
bad += [(r['实例ID'], 'ellipsis') for r in rows if '…' in r['自动化锚点']]
bad += [(r['实例ID'], '状态/载体不一致') for r in rows
        if (r['状态'] == '🔶' and 'WebDriver' not in r['载体'])
        or (r['状态'] == '⛔' and r['载体'].startswith('CI'))]
assert not bad, bad
print('锚点实名与状态-载体一致性校验通过')
EOF
```

CSV 由 `scripts/gen_case_map.py` 生成；改槽位/锚点/缺口请改脚本后重跑，再跑上面两个校验。
